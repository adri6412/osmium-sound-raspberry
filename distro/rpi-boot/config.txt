# Osmium Sound — Raspberry Pi 4 + Pi 5, config.txt (per boot partition A/B).
#
# One copy sits on EACH boot FAT (hifi-boot-a / hifi-boot-b). It is identical
# for both slots: what differs per slot is cmdline.txt (root= and rauc.slot=).
# The VideoCore firmware reads this, then auto-selects the kernel by board
# (kernel8.img on the Pi 4 / BCM2711, kernel_2712.img on the Pi 5 / BCM2712),
# so NO explicit `kernel=` line is needed — raspi-firmware names the images
# exactly as the firmware expects. The initramfs name IS board-specific, so it
# is set under the per-board filters below.
#
# NOTE: [pi4]/[pi5] match the Model B boards ONLY — they do NOT match the
# Compute Modules. The Compute Module 4 (BCM2711, same silicon as the Pi 4) and
# Compute Module 5 (BCM2712, same as the Pi 5) need their own [cm4]/[cm5]
# sections, or the board-specific `initramfs` line never applies and the module
# fails to boot. The rootfs and both kernels/DTBs are shared, so only this file
# differs per board.

arm_64bit=1
arm_boost=1
disable_overscan=1
# No rainbow splash; plymouth owns the screen from early boot.
disable_splash=1
# Serial console on the GPIO header for headless bring-up; free in production.
enable_uart=1

# CMA is sized HERE (cma-256 on the KMS overlay), NOT via `cma=` on the kernel
# command line. On the Pi 5 / CM5 (BCM2712) a `cma=` arg bypasses the device
# tree `linux,cma` node and its `alloc-ranges`, so CMA lands above 1 GB; the
# VideoCore firmware mailbox can only reach memory below 1 GB, so every mailbox
# request then fails -> `raspberrypi-clk ... failed -22`, no cpufreq, RP1/PCIe
# never probe, and the kernel hangs early (raspberrypi/linux#7230). Sizing CMA
# through the overlay keeps the DT reservation and its <1 GB constraint.

[pi4]
# Pi 4 (BCM2711): KMS driver + the matching initramfs raspi-firmware produced.
dtoverlay=vc4-kms-v3d,cma-256
max_framebuffers=2
initramfs initramfs8 followkernel

[cm4]
# Compute Module 4 (BCM2711): identical kernel/initramfs to the Pi 4.
dtoverlay=vc4-kms-v3d,cma-256
max_framebuffers=2
initramfs initramfs8 followkernel
# The CM4 exposes a SINGLE USB 2.0 line and leaves it in peripheral mode by
# default. On the CM5 IO Board that line is wired only to the USB-C/OTG port
# (the board's USB-A ports are USB 3.0, which the CM4 cannot drive), so force
# host mode to use a USB DAC via the USB-C port + a powered hub.
dtoverlay=dwc2,dr_mode=host

[pi5]
# Pi 5 (BCM2712 / RP1): the Pi-5 KMS overlay + its initramfs. cma-256 here, not
# cma= on cmdline (see the note above the [pi4] block).
dtoverlay=vc4-kms-v3d-pi5,cma-256
max_framebuffers=2
initramfs initramfs_2712 followkernel

[cm5]
# Compute Module 5 (BCM2712 / RP1): identical kernel/initramfs to the Pi 5.
# USB-A on the CM5 IO Board is native USB 3.0 from the module, so no USB overlay
# is needed here.
dtoverlay=vc4-kms-v3d-pi5,cma-256
max_framebuffers=2
initramfs initramfs_2712 followkernel

[all]
# On-board analogue/HDMI audio off: this is a source for USB/I2S DACs. A USB DAC
# needs no overlay; for an I2S HAT, uncomment exactly one line below.
dtparam=audio=off
